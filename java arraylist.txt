import java.io.*;
import java.util.*;

public class Solution {

    public static void main(String[] args) {
        Scanner scan = new Scanner(System.in);
        
        // Read the number of lines
        int n = scan.nextInt();
        
        // Create a 2D ArrayList to store the integers for each line
        ArrayList<ArrayList<Integer>> lines = new ArrayList<>();
        
        for (int i = 0; i < n; i++) {
            int d = scan.nextInt();
            ArrayList<Integer> row = new ArrayList<>();
            for (int j = 0; j < d; j++) {
                row.add(scan.nextInt());
            }
            lines.add(row);
        }
        
        // Read the number of queries
        int q = scan.nextInt();
        
        for (int i = 0; i < q; i++) {
            int x = scan.nextInt(); // Line number (1-based index)
            int y = scan.nextInt(); // Position number (1-based index)
            
            // Adjust for 0-based indexing used in ArrayLists and handle errors
            try {
                int result = lines.get(x - 1).get(y - 1);
                System.out.println(result);
            } catch (IndexOutOfBoundsException e) {
                System.out.println("ERROR!");
            }
        }
        
        scan.close();
    }
}
