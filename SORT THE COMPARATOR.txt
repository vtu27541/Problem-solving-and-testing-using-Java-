class Solution {
public:
    vector<string> sortPeople(vector<string>& names, vector<int>& heights) {
        int n = names.size();
        
        // Create an index array to keep track of the original positions
        std::vector<int> indices(n);
        std::iota(indices.begin(), indices.end(), 0);
        
        // Sort indices in descending order based on their heights
        std::sort(indices.begin(), indices.end(), [&](int a, int b) {
            return heights[a] > heights[b];
        });
        
        // Build the result vector of names using the sorted indices
        std::vector<std::string> sortedNames(n);
        for (int i = 0; i < n; ++i) {
            sortedNames[i] = names[indices[i]];
        }
        
        return sortedNames;
    }
};
